# Configure one parameterized testsuite API validation scenario.
cmake_minimum_required(VERSION 3.28)
project(testsuite_config_test NONE)
enable_testing()

include("${CMAKE_CURRENT_LIST_DIR}/../../../cmake/testsuite.cmake")

if(SCENARIO STREQUAL "explicit-collections")
	testsuite(config COLLECTIONS cases.cmake)
	if(NOT TEST config/explicit)
		message(FATAL_ERROR "Explicit collection did not register its test node.")
	endif()
elseif(SCENARIO STREQUAL "missing-collection")
	testsuite(config COLLECTIONS missing.cmake)
elseif(SCENARIO STREQUAL "duplicate-collection")
	testsuite(config COLLECTIONS cases.cmake nested/../cases.cmake)
elseif(SCENARIO STREQUAL "unsafe-collection")
	testsuite(config COLLECTIONS ../outside.cmake)
else()
	testsuite(config)
endif()
